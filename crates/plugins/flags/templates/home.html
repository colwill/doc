How a service reads them

A service asks with a token, naming itself and the environment it runs in. Over HTTP it is given everything that applies to it in one call. With OpenFeature it asks for one at a time through the SDK's OFREP provider, the targeting key naming the service, and anything not set for it comes back as the default its code passed.

  • Go

  • TypeScript

HTTP

curl -H "Authorization: Bearer $DOC_TOKEN" \
  "{{ platform_url }}/api/v1/plugins/flags/api/evaluate?service=payments-api&environment=production"

The answer carries an ETag, so polling it costs a 304.

Go

go get github.com/open-feature/go-sdk github.com/open-feature/go-sdk-contrib/providers/ofrep
import (
	"context"
	"os"

	"github.com/open-feature/go-sdk-contrib/providers/ofrep"
	"github.com/open-feature/go-sdk/openfeature"
)

provider := ofrep.NewProvider(
	"{{ platform_url }}/api/v1/plugins/flags/api",
	ofrep.WithBearerToken(os.Getenv("DOC_TOKEN")),
)
if err := openfeature.SetProviderAndWait(provider); err != nil {
	return err
}
openfeature.SetEvaluationContext(openfeature.NewEvaluationContext("payments-api", map[string]any{
	"environment": "production",
}))
client := openfeature.NewDefaultClient()

Rust

cargo add open-feature open-feature-ofrep reqwest
use open_feature::{EvaluationContext, OpenFeature};
use open_feature_ofrep::{OfrepOptions, OfrepProvider};
use reqwest::header::{AUTHORIZATION, HeaderMap, HeaderValue};

let mut headers = HeaderMap::new();
let bearer = format!("Bearer {}", std::env::var("DOC_TOKEN")?);
headers.insert(AUTHORIZATION, HeaderValue::from_str(&bearer)?);
let provider = OfrepProvider::new(OfrepOptions {
    base_url: "{{ platform_url }}/api/v1/plugins/flags/api".to_string(),
    headers,
    ..Default::default()
})
.await?;

let client = {
    let mut api = OpenFeature::singleton_mut().await;
    api.set_provider(provider).await;
    api.set_evaluation_context(
        EvaluationContext::default()
            .with_targeting_key("payments-api")
            .with_custom_field("environment", "production"),
    )
    .await;
    api.create_client()
};

{% if role == "flag" %}let new_checkout = client.get_bool_value("new-checkout", None, None).await.unwrap_or(false);{% else %}let log_level = client.get_string_value("log-level", None, None).await.unwrap_or_else(|_| "info".into());{% endif %}

TypeScript

npm install @openfeature/server-sdk @openfeature/ofrep-provider
import { OpenFeature } from '@openfeature/server-sdk';
import { OFREPProvider } from '@openfeature/ofrep-provider';

await OpenFeature.setProviderAndWait(
  new OFREPProvider({
    baseUrl: '{{ platform_url }}/api/v1/plugins/flags/api',
    headers: [['Authorization', `Bearer ${process.env.DOC_TOKEN}`]],
  }),
);
OpenFeature.setContext({ targetingKey: 'payments-api', environment: 'production' });
const client = OpenFeature.getClient();

{% if role == "flag" %}const newCheckout = await client.getBooleanValue('new-checkout', false);{% else %}const logLevel = await client.getStringValue('log-level', 'info');{% endif %}
